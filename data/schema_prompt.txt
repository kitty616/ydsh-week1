-- 数据覆盖区间：2024-01-01 ~ 2024-12-31（2024 年历史快照，非实时数据）

-- ydsh_ads.ads_kpi_core（核心 KPI 指标长表，每个指标一行，含口径说明）
   metric_code string  // 指标编码，如 KPI_GMV,
   metric_name string  // 指标中文名，如 GMV 成交总额,
   metric_value double  // 指标值（数值）,
   unit string  // 单位,
   calc_rule string  // 计算口径说明

-- ydsh_ads.ads_sales_daily_report（销售日报，每天一行，含环比与 7 日移动平均）
   dt string  // 日期 yyyy-MM-dd,
   gmv double  // 当日成交总额（元）,
   order_cnt bigint  // 当日订单量,
   user_cnt bigint  // 当日下单用户数,
   aov double  // 当日客单价（元/单）,
   dod_rate double  // GMV 日环比（小数）,
   wow_rate double  // GMV 周同比（对 7 天前，小数）,
   ma7_gmv double  // 近 7 日 GMV 移动平均

-- ydsh_ads.ads_category_topn（品类销售排名，按全年 GMV 降序）
   rank_no int  // 排名，1 为最高,
   category string  // 一级品类名称,
   gmv double  // 该品类全年 GMV（元）,
   gmv_share double  // 占全站 GMV 比重（小数）,
   cum_share double  // 累计占比，帕累托分析用,
   order_cnt bigint  // 关联订单数,
   item_qty bigint  // 销售件数,
   avg_item_price double  // 件单价（元/件）

-- ydsh_ads.ads_user_segment（用户价值分层汇总（基于 RFM 的价值分层，分层名如「重要价值用户」，**不是会员等级**；问会员等级要用 dim_user GROUP BY member_level_name））
   user_segment string  // 用户分层名称，如 重要价值用户,
   user_cnt bigint  // 该层用户数,
   user_share double  // 用户占比,
   gmv double  // 该层 GMV,
   gmv_share double  // GMV 占比,
   avg_gmv double  // 人均 GMV,
   avg_order_cnt double  // 人均订单数

-- ydsh_ads.ads_monthly_trend（月度经营趋势）
   month string  // 月份 yyyy-MM,
   month_seq int  // 月份序号 1-12,
   gmv double  // 当月 GMV,
   order_cnt bigint  // 当月订单量,
   user_cnt bigint  // 当月下单用户数,
   aov double  // 当月客单价,
   mom_rate double  // GMV 环比（小数）,
   gmv_share double  // 占全年 GMV 比重

-- ydsh_ads.ads_region_sales（省份销售分布，按用户所在省份汇总，15 个省合计等于全站 GMV）
   province string  // 省份全称，带「省」后缀（如 广东省、上海市）；过滤用 LIKE '%广东%'，不要写等号精确匹配,
   order_cnt bigint  // 订单量,
   user_cnt bigint  // 下单用户数,
   gmv double  // 成交额（元）,
   aov double  // 客单价（元/单）,
   gmv_ratio double  // GMV 占比（百分数，6.89 表示 6.89%）

-- ydsh_ads.ads_channel_sales（渠道销售分布，4 个渠道合计等于全站 GMV）
   channel_code string  // 渠道编码,
   channel_name string  // 渠道名称,
   order_cnt bigint  // 订单量,
   gmv double  // 成交额（元）,
   aov double  // 客单价（元/单）,
   gmv_ratio double  // GMV 占比（百分数）

-- ydsh_ads.ads_store_rank（门店 GMV 排名，**只含前 15 家门店**，不是全量门店）
   rank_no int  // 排名，1 为最高,
   store_id string  // 门店编号,
   store_name string  // 门店名称,
   province string  // 门店所在省份，带「省/市」后缀,
   gmv double  // 该门店全年 GMV（元）,
   order_cnt bigint  // 订单量,
   aov double  // 客单价（元/单）

-- ydsh_ads.ads_kpi_overview（核心指标宽表，一行一个指标共 8 行，适合「大盘怎么样」这类概览）
   kpi_name string  // 指标名称,
   kpi_value double  // 指标值,
   unit string  // 单位,
   calc_rule string  // 计算口径说明

-- ydsh_dws.dws_trade_day（天粒度全站交易汇总，按日期分区表；**没有 store_id / province / city**，按城市或门店看销售请用 dws_store_day）
   dt string  // 日期 yyyy-MM-dd（与 order_date 同值）,
   gmv double  // 成交总额,
   order_cnt bigint  // 订单量,
   user_cnt bigint  // 下单用户数,
   store_cnt bigint  // 有成交门店数,
   discount_sum double  // 优惠金额合计,
   cancel_cnt bigint  // 取消订单数,
   aov double  // 客单价

-- ydsh_dws.dws_trade_month（月粒度交易汇总，只有 month 一个月度列，没有门店/省份维度）
   month string  // 月份 yyyy-MM,
   gmv double  // 成交总额,
   order_cnt bigint  // 订单量,
   user_cnt bigint  // 下单用户数,
   aov double  // 客单价,
   cancel_cnt bigint  // 取消订单数,
   mom_rate double  // GMV 环比

-- ydsh_dws.dws_category_day（天×品类销售汇总，GMV 已按订单实付金额分摊；dt 是 yyyy-MM-dd，筛整月要写 dt LIKE '2024-06%'。**日粒度**：一个月约 250 行，要「6 月各品类」必须 GROUP BY category 求和，不要 GROUP BY dt）
   dt string  // 日期,
   category string  // 一级品类,
   order_cnt bigint  // 关联订单数,
   item_qty bigint  // 销售件数,
   gmv double  // 分摊后成交金额

-- ydsh_dws.dws_store_day（天×门店销售汇总。**日粒度**：一家店一个月 30 行，问「各门店」要 GROUP BY store_id 求和，不要 GROUP BY dt。汇总时 aov 要重算 SUM(gmv)/SUM(order_cnt)，**不能 SUM(aov)**）
   dt string  // 日期,
   store_id string  // 门店编号,
   gmv double  // 成交总额,
   order_cnt bigint  // 订单量,
   user_cnt bigint  // 下单用户数,
   aov double  // 客单价

-- ydsh_dws.dws_channel_day（天×渠道销售汇总，只有 channel_code 一个渠道维度。**日粒度**：要汇总就 GROUP BY channel_code，不要 GROUP BY dt）
   dt string  // 日期,
   channel_code string  // 渠道编码（关联 dim_channel）,
   gmv double  // 成交总额,
   order_cnt bigint  // 订单量,
   user_cnt bigint  // 下单用户数,
   aov double  // 客单价

-- ydsh_dwd.dwd_order（订单事实表（明细粒度，已清洗），按 order_date 分区）
   order_id string  // 订单号,
   user_id string  // 用户编号,
   store_id string  // 门店编号,
   order_timestamp string  // 下单时间 yyyy-MM-dd HH:mm:ss,
   pay_amount double  // 实付金额（元）,
   discount_amount double  // 优惠金额,
   pay_type_code string  // 支付方式编码,
   order_status_code string  // 订单状态编码（OS04=已取消）,
   channel_code string  // 渠道编码,
   order_cnt int  // 固定为 1，用于计数,
   order_date string  // 订单日期 yyyy-MM-dd，分区字段,
   order_date string  // 订单日期 yyyy-MM-dd，分区字段

-- ydsh_dwd.dim_product（商品维度表）
   product_id string  // 商品编号,
   product_name string,
   category string  // 一级品类,
   sub_category string,
   brand string  // 品牌,
   price double  // 吊牌价,
   cost double,
   stock int,
   status string,
   launch_date string

-- ydsh_dwd.dim_store（门店维度表）
   store_id string  // 门店编号,
   store_name string  // 门店名称,
   province string  // 省份全称，带省/市后缀（如 上海市、江苏省），过滤用 LIKE '%上海%',
   city string  // 城市全称，同样带后缀,
   district string,
   address string,
   open_date string,
   area double,
   manager string

-- ydsh_dwd.dim_channel（渠道维度表）
   channel_code string  // 渠道编码,
   channel_name string  // 渠道名称

-- ydsh_dwd.dim_pay_type（支付方式维度表，2 列。dwd_order.pay_type_code 关联 pay_code 取中文名）
   pay_code string  // 支付方式编码，与 dwd_order.pay_type_code 同值,
   pay_name string  // 支付方式名称，如 微信支付 / 支付宝

-- ydsh_dwd.dim_order_status（订单状态维度表。dwd_order.order_status_code 关联 status_code）
   status_code string  // 状态编码，与 dwd_order.order_status_code 同值，如 OS04,
   status_name string  // 状态名称，如 已取消,
   is_finished int  // 是否已完成状态，1 是 0 否

-- ydsh_dwd.dim_user（用户维度表。province / city 是**用户**所在地区（不是门店），15 省合计等于全站 GMV，口径与 ads_region_sales 一致）
   user_id string  // 用户编号，关联 dwd_order.user_id,
   gender string  // 性别,
   age int  // 年龄,
   province string  // 用户所在省份，带「省/市」后缀，过滤用 LIKE '%广东%',
   city string  // 用户所在城市，带后缀,
   region string  // 用户所在大区,
   register_date string  // 注册日期 yyyy-MM-dd,
   member_level_code string  // 会员等级编码,
   member_level_name string  // 会员等级名称
